#view completo
import asyncio
import os
import time

SCREENSHOT_DIR = "erros_automacao"

def log(msg):
    print(f"[TABELA] {msg}", flush=True)

async def _screenshot(page, label):
    try:
        os.makedirs(SCREENSHOT_DIR, exist_ok=True)
        path = f"{SCREENSHOT_DIR}/erro_{label.replace(' ', '_')}_{int(time.time())}.png"
        await page.screenshot(path=path, full_page=True)
    except Exception:
        pass

# ----------------------------------------------------------------------
# 1. VIEW - PIS e COFINS
# ----------------------------------------------------------------------
async def preencher_tabela_view_pis_cofins(page, dados, log_callback):
    log_callback("⚙️ [1/4] Iniciando preenchimento VIEW - PIS e COFINS...")
    try:
        view_nome = dados.get("pis_cofins", {}).get("view_nome", "")
        view_sql = dados.get("pis_cofins", {}).get("view_sql", "")

        btn_expandir = page.locator("xpath=/html/body/div[4]/div[1]/div[1]/div[1]/div[1]/div[2]/div[1]/div[1]/div[1]/div[2]/div[1]/div[1]/div[1]/div[2]/div[1]/div[1]/div[1]/div[1]/span[1]").first
        if not await btn_expandir.is_visible():
            btn_expandir = page.locator("div.avatar, span.avatar-content").nth(0)

        log_callback("   👉 Expandindo PIS/COFINS...")
        await btn_expandir.click(force=True)
        await page.wait_for_timeout(1200)

        form_aberto = page.locator("form").first

        if view_nome:
            log_callback(f"   ✏️ Preenchendo Nome: {view_nome}")
            input_nome = page.locator("xpath=/html/body/div[4]/div[1]/div[1]/div[1]/div[1]/div[2]/div[1]/div[1]/div[1]/div[2]/div[1]/div[1]/div[1]/div[3]/div[1]/div[1]/div[1]/form[1]/div[1]/input[1]").first
            if not await input_nome.is_visible():
                input_nome = form_aberto.locator("input").first
            await input_nome.click()
            await input_nome.fill("")
            await input_nome.fill(view_nome)
            await input_nome.evaluate("el => { el.dispatchEvent(new Event('input', { bubbles: true })); el.dispatchEvent(new Event('change', { bubbles: true })); }")

        if view_sql:
            log_callback("   ✏️ Preenchendo SQL...")
            textarea_sql = page.locator("xpath=/html/body/div[4]/div[1]/div[1]/div[1]/div[1]/div[2]/div[1]/div[1]/div[1]/div[2]/div[1]/div[1]/div[1]/div[3]/div[1]/div[1]/div[1]/form[1]/div[2]/textarea[1]").first
            if not await textarea_sql.is_visible():
                textarea_sql = form_aberto.locator("textarea").first
            await textarea_sql.click()
            await textarea_sql.fill("")
            await textarea_sql.fill(view_sql)
            await textarea_sql.evaluate("el => { el.dispatchEvent(new Event('input', { bubbles: true })); el.dispatchEvent(new Event('change', { bubbles: true })); }")

        log_callback("   💾 Salvar PIS/COFINS...")
        btn_salvar = page.locator("xpath=/html/body/div[4]/div[1]/div[1]/div[1]/div[1]/div[2]/div[1]/div[1]/div[1]/div[2]/div[1]/div[1]/div[1]/div[3]/div[1]/div[1]/div[1]/form[1]/div[3]/button[1]").first
        if not await btn_salvar.is_visible():
            btn_salvar = page.locator("button:has-text('Salvar')").first
        await btn_salvar.scroll_into_view_if_needed()
        await btn_salvar.click()
        await page.wait_for_timeout(1500)

        log_callback("   🔽 Recolhendo card PIS/COFINS...")
        await btn_expandir.click(force=True)
        await page.wait_for_timeout(1000)

        log_callback("✅ VIEW - PIS e COFINS finalizada!")
        return True
    except Exception as e:
        log_callback(f"❌ Erro em VIEW PIS/COFINS: {e}")
        await _screenshot(page, "erro_pis_cofins")
        return False

# ORQUESTRADOR PRINCIPAL
# ----------------------------------------------------------------------
async def preencher_tabelas_mix(page, template_dados, log_callback):
    """Executa as 4 VIEWs em ordem sequencial."""
    r1 = await preencher_tabela_view_pis_cofins(page, template_dados, log_callback)
    r2 = await preencher_tabela_view_icms_saida(page, template_dados, log_callback)
    r3 = await preencher_tabela_view_icms_entrada(page, template_dados, log_callback)
    r4 = await preencher_tabela_view_ibs_cbs(page, template_dados, log_callback)

    if all([r1, r2, r3, r4]):
        log_callback("🎉 TODAS AS VIEWS FORAM PREENCHIDAS E SALVAS COM SUCESSO!")
        return True
    else:
        log_callback("⚠️ Ocorreu falha em uma ou mais VIEWs durante a execução.")
        return False
